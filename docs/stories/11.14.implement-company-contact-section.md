### Component Implementation

**File**: `web/components/ipo-detail/CompanyContactCard.tsx`

```typescript
import React from 'react';
import { Phone, Mail, MapPin, ExternalLink } from 'lucide-react';
import type { IPODetail } from '@/lib/db/types';

interface CompanyContactCardProps {
  ipoDetails: IPODetail;
  companyName: string;
  website?: string | null;
}

export function CompanyContactCard({ ipoDetails, companyName, website }: CompanyContactCardProps) {
  // Check if any contact information exists
  const hasContactInfo =
    ipoDetails.companyAddress ||
    ipoDetails.companyPhone ||
    ipoDetails.companyEmail ||
    ipoDetails.complianceOfficer;

  if (!hasContactInfo) {
    return (
      <div className="rounded-lg border border-gray-200 bg-gray-50 p-6">
        <p className="text-sm text-gray-500">Contact information not available</p>
      </div>
    );
  }

  // Format address
  const fullAddress = [
    ipoDetails.companyAddress,
    ipoDetails.companyCity,
    ipoDetails.companyState,
    ipoDetails.companyPincode
  ].filter(Boolean).join(', ');

  return (
    <div className="rounded-lg border border-gray-200 bg-white p-6 shadow-sm">
      <h3 className="mb-4 text-lg font-semibold text-gray-900">
        Company Contact Information
      </h3>

      {/* Company Details Section */}
      <div className="space-y-4">
        {/* Company Name */}
        <div>
          <p className="text-sm font-medium text-gray-700">Company Name</p>
          <p className="text-base text-gray-900">{companyName}</p>
        </div>

        {/* Registered Office Address */}
        {ipoDetails.companyAddress && (
          <div>
            <div className="flex items-start gap-2">
              <MapPin className="mt-1 h-4 w-4 text-gray-500" />
              <div>
                <p className="text-sm font-medium text-gray-700">Registered Office</p>
                <p className="text-sm text-gray-900 whitespace-pre-line">{fullAddress}</p>
                {/* Optional: Google Maps link */}
                <a
                  href={`https://maps.google.com/?q=${encodeURIComponent(fullAddress)}`}
                  target="_blank"
                  rel="noopener noreferrer"
                  className="mt-1 inline-flex items-center gap-1 text-sm text-blue-600 hover:text-blue-700"
                >
                  View on Map <ExternalLink className="h-3 w-3" />
                </a>
              </div>
            </div>
          </div>
        )}

        {/* Phone */}
        {ipoDetails.companyPhone && (
          <div>
            <div className="flex items-center gap-2">
              <Phone className="h-4 w-4 text-gray-500" />
              <div>
                <p className="text-sm font-medium text-gray-700">Phone</p>
                <a
                  href={`tel:${ipoDetails.companyPhone}`}
                  className="text-sm text-blue-600 hover:text-blue-700"
                >
                  {ipoDetails.companyPhone}
                </a>
              </div>
            </div>
          </div>
        )}

        {/* Email */}
        {ipoDetails.companyEmail && (
          <div>
            <div className="flex items-center gap-2">
              <Mail className="h-4 w-4 text-gray-500" />
              <div>
                <p className="text-sm font-medium text-gray-700">Email (Investor Relations)</p>
                <a
                  href={`mailto:${ipoDetails.companyEmail}`}
                  className="text-sm text-blue-600 hover:text-blue-700"
                >
                  {ipoDetails.companyEmail}
                </a>
              </div>
            </div>
          </div>
        )}

        {/* Website */}
        {website && (
          <div>
            <div className="flex items-center gap-2">
              <ExternalLink className="h-4 w-4 text-gray-500" />
              <div>
                <p className="text-sm font-medium text-gray-700">Website</p>
                <a
                  href={website}
                  target="_blank"
                  rel="noopener noreferrer"
                  className="text-sm text-blue-600 hover:text-blue-700"
                >
                  {website}
                </a>
              </div>
            </div>
          </div>
        )}

        {/* Compliance Officer */}
        {ipoDetails.complianceOfficer && (
          <div className="mt-6 border-t border-gray-200 pt-4">
            <p className="mb-2 text-sm font-semibold text-gray-700">Compliance Officer</p>
            <div className="space-y-2">
              <p className="text-sm text-gray-900">{ipoDetails.complianceOfficer}</p>

              {ipoDetails.complianceOfficerPhone && (
                <div className="flex items-center gap-2">
                  <Phone className="h-4 w-4 text-gray-500" />
                  <a
                    href={`tel:${ipoDetails.complianceOfficerPhone}`}
                    className="text-sm text-blue-600 hover:text-blue-700"
                  >
                    {ipoDetails.complianceOfficerPhone}
                  </a>
                </div>
              )}

              {ipoDetails.complianceOfficerEmail && (
                <div className="flex items-center gap-2">
                  <Mail className="h-4 w-4 text-gray-500" />
                  <a
                    href={`mailto:${ipoDetails.complianceOfficerEmail}`}
                    className="text-sm text-blue-600 hover:text-blue-700"
                  >
                    {ipoDetails.complianceOfficerEmail}
                  </a>
                </div>
              )}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
```

## Risks and Mitigation

### Risk 1: Data Availability
- **Issue**: Contact information not available in all DRHPs
- **Mitigation**: Make all fields optional, handle empty state gracefully
- **Fallback**: Display "Contact information not available"

### Risk 2: Phone Format Variations
- **Issue**: International phone formats vary widely
- **Mitigation**: Flexible phone validation, accept various formats
- **Fallback**: Store as text, minimal format validation

### Risk 3: Address Formatting
- **Issue**: Indian addresses have no standard format
- **Mitigation**: Free-text address field, city/state/pincode separate
- **Enhancement**: Future address parser for structured data

### Risk 4: Email Spam
- **Issue**: Public email addresses may receive spam
- **Mitigation**: No scraping required (reduces exposure)
- **Note**: Investor relations emails are publicly available anyway
