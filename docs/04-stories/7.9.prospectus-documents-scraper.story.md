    async validateDocumentUrl(url: string): Promise<ValidationResult> {
      try {
        const response = await fetch(url, {
          method: 'HEAD',
          headers: { 'User-Agent': 'IPODhan/1.0' },
          signal: AbortSignal.timeout(10000), // 10 second timeout
        });

        if (!response.ok) {
          return { valid: false, error: `HTTP ${response.status}` };
        }

        const contentType = response.headers.get('content-type');
        if (!contentType?.includes('application/pdf')) {
          return { valid: false, error: 'Not a PDF' };
        }

        const contentLength = response.headers.get('content-length');
        const fileSize = contentLength ? parseInt(contentLength) : null;

        return { valid: true, fileSize };
      } catch (error) {
        return { valid: false, error: error.message };
      }
    }

### Technical Approach

**Document Matching Challenge:**
Company names on NSE/BSE prospectus pages may differ from company names in our database:
- Database: "Tech Corporation Limited"
- NSE Page: "Tech Corp Ltd"
- Solution: Fuzzy string matching with Levenshtein distance

**Fuzzy Matching Implementation:**
```typescript
// Normalize company names
function normalize(name: string): string {
  return name
    .toLowerCase()
    .replace(/\b(limited|ltd|inc|pvt|private)\b/gi, '')
    .replace(/\s+/g, ' ')
    .trim();
}

// Calculate similarity
function similarity(str1: string, str2: string): number {
  const normalized1 = normalize(str1);
  const normalized2 = normalize(str2);
  // Use Levenshtein distance algorithm
  const distance = levenshtein(normalized1, normalized2);
  const maxLength = Math.max(normalized1.length, normalized2.length);
  return ((maxLength - distance) / maxLength) * 100;
}

// Match document to IPO
if (similarity(docName, ipoName) >= 85) {
  // Match found
}
```

**Foreign Key Cascade:**
- `ON DELETE CASCADE` - When IPO is deleted, automatically delete associated documents (already exists)

**Code References:**
- Import from: `import { documents } from '@/lib/db/schema';`
- Table variable name: `documents` (NOT `ipoDocuments`)
- Schema location: `web/lib/db/schema.ts` (NOT `web/drizzle/schema.ts`)

### Error Handling

**Scraper-Level Errors:**
- Page load failures: Retry 3 times with exponential backoff
- Parse errors: Log and continue processing
- Validation errors: Skip invalid documents, continue with others

**URL Validation Errors:**
- Timeout: Retry 2 times with 5s delay
- Invalid response: Log for manual review, store document without file size
- Non-PDF content: Skip document, log warning

**Database Errors:**
- Duplicate URL: Update existing document (conflict resolution)
- Foreign key violation: Skip document (IPO not found in database)
- Connection errors: Retry entire storage operation

**Logging Strategy:**
```typescript
// Success
this.logger.info({
  scraper: 'prospectus',
  phase: 'complete',
  documentsFound: 75,
  documentsValidated: 70,
  documentsMatched: 65,
  documentsStored: 65,
  executionTime: 45000,
}, 'Prospectus scraper completed successfully');

// Failure
this.logger.error({
  scraper: 'prospectus',
  phase: 'validation',
  url: 'https://example.com/doc.pdf',
  error: 'Timeout after 10s',
}, 'Document validation failed');
```
