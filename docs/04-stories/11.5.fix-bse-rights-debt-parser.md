**Conditional Validation Pattern:**
```typescript
// In scraper/src/utils/validators.ts
export const bseDetailSchema = z.object({
  symbol: z.string().nullable(),
  leadManagers: z.array(z.string()).nullable(),
  category: z.enum(['MAINBOARD', 'SME', 'RIGHTS', 'NCD']),
  openDate: z.string().nullable(),
  closeDate: z.string().nullable(),
  priceRangeMin: z.number(),
  priceRangeMax: z.number(),
  issueSize: z.number(),
  lotSize: z.number(),
  faceValue: z.number(),
  registrar: z.string().nullable(),
  sponsorBanks: z.array(z.string()).nullable(),
  issueShares: z.number(),
}).refine((data) => {
  // For MAINBOARD and SME, symbol and leadManagers are required
  if (data.category === 'MAINBOARD' || data.category === 'SME') {
    if (data.symbol === null) {
      return false; // Validation fails
    }
    if (data.leadManagers === null) {
      return false; // Validation fails
    }
  }
  // For RIGHTS and NCD, symbol and leadManagers are optional
  return true;
}, {
  message: 'Symbol and leadManagers are required for MAINBOARD/SME IPOs',
  path: ['symbol', 'leadManagers'], // Error path for debugging
});

// Usage in scraper
try {
  const validatedData = bseDetailSchema.parse(extractedData);
  // Proceed with database insertion
} catch (error) {
  if (error instanceof z.ZodError) {
    console.error('Validation failed:', error.errors);
    // Log which fields failed and why
  }
}
```
