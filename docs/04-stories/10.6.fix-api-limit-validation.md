### **OPTION 1: Dedicated Calendar Endpoint (RECOMMENDED)**

**Pros**:
- Clean separation of concerns
- No limit restrictions needed
- Optimized query for calendar use case
- Better caching strategy
- Doesn't affect existing API

**Cons**:
- Additional endpoint to maintain
- Slight code duplication

**Implementation**:
```typescript
// app/api/calendar/[category]/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { z } from 'zod';

const paramsSchema = z.object({
  category: z.enum(['MAINBOARD', 'SME'])
});

const querySchema = z.object({
  year: z.coerce.number().int().optional(),
  startDate: z.string().optional(),
  endDate: z.string().optional(),
});

export async function GET(
  request: NextRequest,
  { params }: { params: { category: string } }
) {
  try {
    // Validate params
    const { category } = paramsSchema.parse(params);
    const searchParams = Object.fromEntries(request.nextUrl.searchParams);
    const { year, startDate, endDate } = querySchema.parse(searchParams);

    // Build query - NO LIMIT
    let query = `
      SELECT * FROM ipos
      WHERE category = $1
    `;
    const queryParams: any[] = [category];

    if (year) {
      query += ` AND EXTRACT(YEAR FROM open_date) = $${queryParams.length + 1}`;
      queryParams.push(year);
    }

    if (startDate && endDate) {
      query += ` AND open_date BETWEEN $${queryParams.length + 1} AND $${queryParams.length + 2}`;
      queryParams.push(startDate, endDate);
    }

    query += ` ORDER BY open_date ASC`;

    const result = await db.query(query, queryParams);

    return NextResponse.json({
      ipos: result.rows,
      count: result.rows.length,
      category,
    });

  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Invalid parameters', details: error.errors },
        { status: 400 }
      );
    }
    console.error('Calendar API error:', error);
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    );
  }
}

// Cache for 5 minutes
export const revalidate = 300;
```

```typescript
// app/mainboard-ipo-calendar/page.tsx
async function fetchAllIpos(category: string): Promise<IPO[]> {
  const allIpos: IPO[] = [];
  let page = 1;
  let hasMore = true;

  while (hasMore) {
    const response = await fetch(
      `/api/ipos?category=${category}&limit=100&page=${page}`
    );

    if (!response.ok) {
      throw new Error('Failed to fetch IPOs');
    }

    const data = await response.json();
    allIpos.push(...data.items);

    hasMore = data.pagination.hasNextPage;
    page++;

    // Safety limit to prevent infinite loops
    if (page > 50) break;
  }

  return allIpos;
}

export default async function MainboardCalendarPage() {
  const ipos = await fetchAllIpos('MAINBOARD');
  return <CalendarView ipos={ipos} category="MAINBOARD" />;
}
```

// app/api/ipos/route.ts
import { rateLimit } from '@/lib/middleware/rate-limit';

export async function GET(request: NextRequest) {
  const searchParams = Object.fromEntries(request.nextUrl.searchParams);
  const { limit } = ipoQuerySchema.parse(searchParams);

  // Apply stricter rate limit for large queries
  if (limit > 100) {
    const rateLimitResult = await rateLimit(request, {
      maxRequests: 10,
      windowMs: 60000 // 10 requests per minute

    if (!rateLimitResult.success) {
      return NextResponse.json(

}
